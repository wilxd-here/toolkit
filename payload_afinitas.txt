<script>
    // ANIMASI BACKGROUND
    const canvas = document.getElementById('particles');
    const ctx = canvas.getContext('2d');
    function resize() { canvas.width = window.innerWidth; canvas.height = window.innerHeight; }
    resize(); window.addEventListener('resize', resize);
    const particles = [];
    for (let i = 0; i < 40; i++) {
        particles.push({
            x: Math.random() * canvas.width, y: Math.random() * canvas.height,
            size: Math.random() * 2 + 0.5, speedX: (Math.random() - 0.5) * 0.5, speedY: (Math.random() - 0.5) * 0.5,
            opacity: Math.random() * 0.5 + 0.1, color: Math.random() > 0.5 ? '#8a5aff' : '#ff8a5a'
        });
    }
    function animate() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        particles.forEach(p => {
            p.x += p.speedX; p.y += p.speedY;
            if (p.x < 0) p.x = canvas.width; if (p.x > canvas.width) p.x = 0;
            if (p.y < 0) p.y = canvas.height; if (p.y > canvas.height) p.y = 0;
            ctx.beginPath(); ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
            ctx.fillStyle = p.color; ctx.globalAlpha = p.opacity; ctx.fill();
        });
        ctx.globalAlpha = 1;
        requestAnimationFrame(animate);
    }
    animate();

    // FUNGSI UPLOAD GAMBAR KE NEXADEV
    async function uploadToNexa(file) {
        const formData = new FormData();
        formData.append("files[]", file);
        const response = await fetch("https://api.nexadev.my.id/uploder/", {
            method: "POST",
            body: formData
        });
        const data = await response.json();
        if (data.success && data.files && data.files[0].url) {
            return data.files[0].url;
        }
        throw new Error("Gagal upload gambar.");
    }

    // LOGIKA UTAMA GENERATOR AFINITAS ML
    async function generateAfinitas() {
        const type = document.getElementById('mlType').value;
        const img1Input = document.getElementById('img1');
        const img2Input = document.getElementById('img2');
        const name1 = document.getElementById('name1').value.trim();
        const name2 = document.getElementById('name2').value.trim();

        // Validasi
        if (!name1 || !name2) {
            alert("Harap isi nickname kedua pemain!");
            return;
        }
        if (img1Input.files.length === 0 || img2Input.files.length === 0) {
            alert("Harap upload foto profil untuk kedua pemain!");
            return;
        }

        const btn = document.getElementById('btnGenerate');
        const loader = document.getElementById('loader');
        const placeholder = document.getElementById('placeholder');
        const resultImg = document.getElementById('resultImg');
        const btnDownload = document.getElementById('btnDownload');

        btn.disabled = true;
        placeholder.style.display = "none";
        resultImg.style.display = "none";
        btnDownload.style.display = "none";
        loader.style.display = "block";

        try {
            btn.innerText = "Mengunggah 2 Foto...";
            
            // Upload kedua gambar secara paralel agar lebih cepat
            const [urlImg1, urlImg2] = await Promise.all([
                uploadToNexa(img1Input.files[0]),
                uploadToNexa(img2Input.files[0])
            ]);

            btn.innerText = "Memproses Afinitas ML...";

            // SUSUN URL API ENDPOINT
            // Catatan: Jika nama parameter di afinitasml.cjs berbeda, ubah teks name1, name2, pp1, pp2 di bawah ini.
            const apiUrl = `https://apii.nexadev.my.id/afinitasml?type=${encodeURIComponent(type)}&name1=${encodeURIComponent(name1)}&name2=${encodeURIComponent(name2)}&pp1=${encodeURIComponent(urlImg1)}&pp2=${encodeURIComponent(urlImg2)}`;

            // Load gambar hasil
            resultImg.onload = function() {
                loader.style.display = "none";
                resultImg.style.display = "block";
                btnDownload.style.display = "inline-block";
                btnDownload.href = apiUrl; 
                
                btn.disabled = false;
                btn.innerText = "Buat Ulang Screenshot";
            };

            resultImg.onerror = function() {
                throw new Error("Gagal merender gambar dari API");
            };

            // Pasang src ke img tag
            resultImg.src = apiUrl + "&_t=" + Date.now();

        } catch (error) {
            console.error(error);
            alert("Terjadi kesalahan. Pastikan server API hidup atau koneksi stabil.");
            
            loader.style.display = "none";
            placeholder.style.display = "block";
            placeholder.innerHTML = "<span style='color:#ff5a8a'>Gagal memproses gambar.</span>";
            btn.disabled = false;
            btn.innerText = "Coba Lagi";
        }
    }
</script>
