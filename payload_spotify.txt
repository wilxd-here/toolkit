<script>
    const urlInput = document.getElementById('urlInput');
    const fetchBtn = document.getElementById('fetchBtn');
    const clearBtn = document.getElementById('clearBtn');
    const logBox = document.getElementById('logBox');
    const resultBox = document.getElementById('resultBox');
    const videoInfo = document.getElementById('videoInfo');
    const downloadOptions = document.getElementById('downloadOptions');

    function addLog(msg, type = 'info') {
        const line = document.createElement('div');
        line.className = 'line ' + type;
        const time = new Date().toLocaleTimeString('id-ID', { hour12: false });
        line.textContent = `[${time}] ${msg}`;
        logBox.appendChild(line);
        logBox.scrollTop = logBox.scrollHeight;
    }

    async function handleDownload() {
        let url = urlInput.value.trim();
        if (!url) { alert('Masukkan link Spotify terlebih dahulu!'); return; }

        fetchBtn.disabled = true;
        fetchBtn.textContent = '⏳ Mengambil...';
        logBox.classList.add('show');
        logBox.innerHTML = '';
        resultBox.classList.remove('show');

        addLog('🔥 Menghubungkan ke API...', 'info');
        addLog(`🔗 URL: ${url}`, 'info');

        try {
            // Menggunakan endpoint API Spotify yang diberikan.
            // Ditambahkan corsproxy.io untuk mencegah potensi block CORS seperti masalah sebelumnya.
            const apiUrl = `https://corsproxy.io/?https://api.nexaadev.my.id/api/spotify?url=${encodeURIComponent(url)}`;
            
            const res = await fetch(apiUrl);
            if (!res.ok) throw new Error(`HTTP error! status: ${res.status}`);

            const json = await res.json();
            addLog('✅ Data berhasil diterima!', 'success');

            // Asumsi struktur data: json.data atau json (sesuaikan jika perlu setelah testing)
            const data = json.data || json; 

            if (!data) {
                throw new Error('Respons API kosong atau format tidak dikenali.');
            }

            // Mencoba mengambil data lagu (asumsi umum struktur respons API Spotify Downloader)
            const title = data.title || data.name || 'Lagu Spotify';
            const artist = data.artist || data.artists || 'Artis Tidak Diketahui';
            const thumb = data.thumbnail || data.cover || data.image || '';
            const downloadUrl = data.download || data.url || data.link;

            // Render Info Lagu
            videoInfo.innerHTML = `
                ${thumb ? `<img src="${thumb}" alt="Cover">` : ''}
                <div class="video-details">
                    <h3>${title}</h3>
                    <p>👤 ${artist}</p>
                </div>
            `;

            // Render Tombol Download
            let buttonsHtml = '';
            
            if (downloadUrl) {
                 buttonsHtml += `
                    <a href="${downloadUrl}" class="dl-btn" target="_blank" rel="noopener">
                        🎵 Download MP3
                        <span>🔗 Unduh Lagu</span>
                    </a>
                `;
            } else {
                 throw new Error('Link download tidak ditemukan pada respons API.');
            }

            downloadOptions.innerHTML = buttonsHtml;
            resultBox.classList.add('show');
            addLog('✨ Selesai! Silakan unduh lagunya.', 'success');

        } catch (e) {
            addLog(`❌ Error: ${e.message}`, 'error');
            alert('Gagal mengambil data lagu. Pastikan link Spotify valid.');
        } finally {
            fetchBtn.disabled = false;
            fetchBtn.textContent = '🚀 Proses Lagu';
        }
    }

    clearBtn.addEventListener('click', () => {
        urlInput.value = '';
        logBox.innerHTML = '';
        logBox.classList.remove('show');
        resultBox.classList.remove('show');
    });

    fetchBtn.addEventListener('click', handleDownload);
    urlInput.addEventListener('keydown', e => {
        if (e.key === 'Enter') handleDownload();
    });

    // PARTICLE BACKGROUND ANIMATION
    const canvas = document.getElementById('particles');
    const ctx = canvas.getContext('2d');

    function resize() {
        canvas.width = window.innerWidth;
        canvas.height = window.innerHeight;
    }
    resize();
    window.addEventListener('resize', resize);

    const particles = [];
    for (let i = 0; i < 40; i++) {
        particles.push({
            x: Math.random() * canvas.width,
            y: Math.random() * canvas.height,
            size: Math.random() * 2 + 0.5,
            speedX: (Math.random() - 0.5) * 0.5,
            speedY: (Math.random() - 0.5) * 0.5,
            opacity: Math.random() * 0.5 + 0.1,
            // Mengubah warna partikel agar senada dengan tema Spotify (Hijau dan Ungu)
            color: Math.random() > 0.5 ? '#1db954' : '#8a5aff'
        });
    }

    function animate() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        particles.forEach(p => {
            p.x += p.speedX;
            p.y += p.speedY;
            if (p.x < 0) p.x = canvas.width;
            if (p.x > canvas.width) p.x = 0;
            if (p.y < 0) p.y = canvas.height;
            if (p.y > canvas.height) p.y = 0;
            ctx.beginPath();
            ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
            ctx.fillStyle = p.color;
            ctx.globalAlpha = p.opacity;
            ctx.fill();
        });
        ctx.globalAlpha = 1;
        requestAnimationFrame(animate);
    }
    animate();
</script>
