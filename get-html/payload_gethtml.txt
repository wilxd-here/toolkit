        if (!session || !session.isVVIP || !session.username) {

        }

        if (session.expiresAt && new Date(session.expiresAt) < new Date()) {
            localStorage.removeItem('xaerisoft_vvip_session');
            window.location.href = '../../login/';
            return;
        }
    })();

    function addLog(msg, type = 'info') {
        const line = document.createElement('div');
        line.className = 'line ' + type;
        const time = new Date().toLocaleTimeString('id-ID', { hour12: false });
        line.textContent = `[${time}] ${msg}`;
        logBox.appendChild(line);
        logBox.scrollTop = logBox.scrollHeight;
    }

    // ============================================================
    // FETCH HTML
    // ============================================================
    async function fetchHTML() {
        let url = urlInput.value.trim();
        if (!url) { alert('Masukkan URL dulu!'); return; }
        if (!url.startsWith('http://') && !url.startsWith('https://')) {
            url = 'https://' + url;
        }

        fetchBtn.disabled = true;
        fetchBtn.textContent = '⏳ Mengambil...';
        logBox.classList.add('show');
        logBox.innerHTML = '';
        resultBox.classList.remove('show');
        codeBox.textContent = '';
        statsRow.innerHTML = '';

        addLog('🔥 Memulai...', 'info');
        addLog(`🔗 URL: ${url}`, 'info');

        try {
            // Pake CORS proxy biar bisa fetch dari browser
            const proxies = [
                'https://api.allorigins.win/raw?url=',
                'https://corsproxy.io/?',
                'https://api.codetabs.com/v1/proxy?quest='
            ];

            let html = null;
            let successProxy = null;

            for (const proxy of proxies) {
                try {
                    addLog(`🌐 Mencoba proxy: ${proxy.split('/')[2]}...`, 'info');
                    const proxyUrl = proxy + encodeURIComponent(url);

                    const res = await fetch(proxyUrl);
                    if (!res.ok) {
                        addLog(`⚠️ HTTP ${res.status}`, 'error');
                        continue;
                    }

                    html = await res.text();
                    if (html && html.length > 50) {
                        successProxy = proxy;
                        addLog(`✅ Berhasil via ${proxy.split('/')[2]}`, 'success');
                        break;
                    }
                } catch (e) {
                    addLog(`⚠️ Gagal: ${e.message}`, 'error');
                }
            }

            if (!html) {
                throw new Error('Semua proxy gagal');
            }

            // Batasi ukuran
            if (html.length > 2 * 1024 * 1024) {
                html = html.substring(0, 2 * 1024 * 1024);
                addLog('⚠️ HTML dipotong (max 2MB)', 'info');
            }

            addLog(`📄 HTML: ${(html.length / 1024).toFixed(1)} KB`, 'success');

            // Parse CSS & JS
            addLog('🎨 Mencari CSS...', 'info');
            const cssUrls = [];

            const cssRegex1 = /<link[^>]+rel=["']stylesheet["'][^>]*href=["']([^"']+)["'][^>]*>/gi;
            const cssRegex2 = /<link[^>]+href=["']([^"']+)["'][^>]*rel=["']stylesheet["'][^>]*>/gi;

            let m;
            while ((m = cssRegex1.exec(html)) !== null) cssUrls.push(m[1]);
            while ((m = cssRegex2.exec(html)) !== null) cssUrls.push(m[1]);

            addLog(`📊 Ketemu ${cssUrls.length} CSS`, 'info');

            // Fetch CSS
            const cssFiles = [];
            for (const cssUrl of [...new Set(cssUrls)].slice(0, 10)) {
                try {
                    const abs = new URL(cssUrl, url).href;
                    const res = await fetch(proxies[0] + encodeURIComponent(abs));
                    const content = await res.text();
                    cssFiles.push({ url: abs, content: content, size: content.length });
                    addLog(`✅ CSS: ${abs.split('/').pop()} (${(content.length / 1024).toFixed(1)} KB)`, 'success');
                } catch (e) {
                    addLog(`⚠️ CSS gagal: ${cssUrl}`, 'error');
                }
            }

            // Parse JS
            addLog('⚡ Mencari JS...', 'info');
            const jsUrls = [];
            const jsRegex = /<script[^>]+src=["']([^"']+)["'][^>]*>/gi;
            while ((m = jsRegex.exec(html)) !== null) jsUrls.push(m[1]);

            addLog(`📊 Ketemu ${jsUrls.length} JS`, 'info');

            // Fetch JS
            const jsFiles = [];
            for (const jsUrl of [...new Set(jsUrls)].slice(0, 10)) {
                try {
                    const abs = new URL(jsUrl, url).href;
                    const res = await fetch(proxies[0] + encodeURIComponent(abs));
                    const content = await res.text();
                    jsFiles.push({ url: abs, content: content, size: content.length });
                    addLog(`✅ JS: ${abs.split('/').pop()} (${(content.length / 1024).toFixed(1)} KB)`, 'success');
                } catch (e) {
                    addLog(`⚠️ JS gagal: ${jsUrl}`, 'error');
                }
            }
       // Gabungin ALL
            let allContent = `/* ============================================\n   🔥 XAERISOFT — ALL ASSETS\n   URL: ${url}\n   HTML: ${html.length} bytes\n   CSS: ${cssFiles.length} file\n   JS: ${jsFiles.length} file\n============================================ */\n\n`;
            allContent += '<!-- ==================== HTML ==================== -->\n';
            allContent += html + '\n\n';

            cssFiles.forEach((f, i) => {
                allContent += `/* ==================== CSS #${i+1} — ${f.url} ==================== */\n`;
                allContent += f.content + '\n\n';
            });

            jsFiles.forEach((f, i) => {
                allContent += `// ==================== JS #${i+1} — ${f.url} ====================\n`;
                allContent += f.content + '\n\n';
            });

            // Simpen state
            state.html = html;
            state.css = cssFiles;
            state.js = jsFiles;
            state.all = allContent;
            state.activeTab = 'html';

            // Stats
            statsRow.innerHTML = `
                <div class="stat-pill">📄 HTML: ${(html.length / 1024).toFixed(1)} KB</div>
                <div class="stat-pill">🎨 CSS: ${cssFiles.length} file</div>
                <div class="stat-pill">⚡ JS: ${jsFiles.length} file</div>
                <div class="stat-pill">✅ via ${successProxy.split('/')[2]}</div>
            `;

            // Tampilkan
            resultBox.classList.add('show');
            switchTab('html');

            addLog('✅ SELESAI!', 'success');

        } catch (e) {
            addLog(`❌ Error: ${e.message}`, 'error');
        } finally {
            fetchBtn.disabled = false;
            fetchBtn.textContent = '🚀 Ambil HTML';
        }
    }

    // ============================================================
    // SWITCH TAB
    // ============================================================
    function switchTab(tab) {
        state.activeTab = tab;
        document.querySelectorAll('.tab-btn').forEach(btn => {
            btn.classList.toggle('active', btn.dataset.tab === tab);
        });

        let content = '';
        let label = '';

        if (tab === 'html') {
            content = state.html;
            label = '📄 HTML SOURCE';
            previewBtn.style.display = 'inline-block';
        } else if (tab === 'css') {
            if (state.css.length === 0) {
                content = '/* Tidak ada CSS eksternal */';
            } else {
                content = state.css.map((f, i) => `/* ===== FILE ${i+1}: ${f.url} ===== */\n${f.content}\n\n`).join('');
            }
            label = '🎨 CSS FILES';
            previewBtn.style.display = 'none';
        } else if (tab === 'js') {
            if (state.js.length === 0) {
                content = '// Tidak ada JS eksternal';
            } else {
                content = state.js.map((f, i) => `// ===== FILE ${i+1}: ${f.url} =====\n${f.content}\n\n`).join('');
            }
            label = '⚡ JS FILES';
            previewBtn.style.display = 'none';
        } else if (tab === 'all') {
            content = state.all;
            label = '📦 ALL ASSETS';
            previewBtn.style.display = 'inline-block';
        }

        resultLabel.textContent = label;
        codeBox.textContent = content || '(kosong)';
    }

            document.execCommand('copy');
            copyBtn.textContent = '✅ Copied!';
            setTimeout(() => copyBtn.textContent = '📋 Copy', 1500);
        });
    });

            color: Math.random() > 0.5 ? '#8a5aff' : '#ff8a5a'
        });
    }

    function animate() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        particles.forEach(p => {
            p.x += p.speedX;
            p.y += p.speedY;
            if (p.x < 0) p.x = canvas.width;
            if (p.x > canvas.width) p.x = 0;
            if (p.y < 0) p.y = canvas.height;
            if (p.y > canvas.height) p.y = 0;
            ctx.beginPath();
            ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
            ctx.fillStyle = p.color;
            ctx.globalAlpha = p.opacity;
            ctx.fill();
        });
        ctx.globalAlpha = 1;
        requestAnimationFrame(animate);
    }
    animate();

    console.log('📥 XAERISOFT GET HTML READY');
</script>
