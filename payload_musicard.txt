<script>
    // ANIMASI BACKGROUND
    const canvas = document.getElementById('particles');
    const ctx = canvas.getContext('2d');
    function resize() { canvas.width = window.innerWidth; canvas.height = window.innerHeight; }
    resize(); window.addEventListener('resize', resize);

    const particles = [];
    for (let i = 0; i < 40; i++) {
        particles.push({
            x: Math.random() * canvas.width, y: Math.random() * canvas.height,
            size: Math.random() * 2 + 0.5, speedX: (Math.random() - 0.5) * 0.5, speedY: (Math.random() - 0.5) * 0.5,
            opacity: Math.random() * 0.5 + 0.1, color: Math.random() > 0.5 ? '#8a5aff' : '#ff8a5a'
        });
    }

    function animate() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        particles.forEach(p => {
            p.x += p.speedX; p.y += p.speedY;
            if (p.x < 0) p.x = canvas.width; if (p.x > canvas.width) p.x = 0;
            if (p.y < 0) p.y = canvas.height; if (p.y > canvas.height) p.y = 0;
            ctx.beginPath(); ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
            ctx.fillStyle = p.color; ctx.globalAlpha = p.opacity; ctx.fill();
        });
        ctx.globalAlpha = 1;
        requestAnimationFrame(animate);
    }
    animate();

    // LOGIKA GENERATOR MUSIC CARD
    async function generateMusicCard() {
        const title = document.getElementById('musicTitle').value.trim();
        const artist = document.getElementById('musicArtist').value.trim();
        const fileInput = document.getElementById('musicCover');

        if (!title || !artist) {
            alert("Harap isi Judul Lagu dan Nama Artis!");
            return;
        }

        // Referensi UI
        const btn = document.getElementById('btnGenerate');
        const loader = document.getElementById('loader');
        const placeholder = document.getElementById('placeholder');
        const resultImg = document.getElementById('resultImg');
        const btnDownload = document.getElementById('btnDownload');

        // Loading state
        btn.disabled = true;
        placeholder.style.display = "none";
        resultImg.style.display = "none";
        btnDownload.style.display = "none";
        loader.style.display = "block";

        let finalImageUrl = ""; // Akan diisi URL dari Nexa Uploader

        // PROSES UPLOAD KE NEXA UPLODER JIKA ADA FILE
        if (fileInput.files.length > 0) {
            btn.innerText = "Mengunggah Cover...";
            const file = fileInput.files[0];
            const formData = new FormData();
            formData.append("files[]", file);

            try {
                // Menggunakan endpoint Nexadev yang sama dengan script.js kamu
                const uploadResponse = await fetch("https://api.nexadev.my.id/uploder/", {
                    method: "POST",
                    body: formData
                });
                
                const uploadData = await uploadResponse.json();
                
                if (uploadData.success && uploadData.files && uploadData.files[0].url) {
                    finalImageUrl = uploadData.files[0].url; 
                } else {
                    throw new Error("Gagal mengunggah foto ke server.");
                }
            } catch (error) {
                console.error(error);
                alert("Gagal mengunggah gambar. Pastikan file valid atau coba beberapa saat lagi.");
                
                // Reset loading
                loader.style.display = "none";
                placeholder.style.display = "block";
                btn.disabled = false;
                btn.innerText = "Generate Card";
                return;
            }
        } else {
            alert("Harap upload gambar cover terlebih dahulu!");
            loader.style.display = "none";
            placeholder.style.display = "block";
            btn.disabled = false;
            btn.innerText = "Generate Card";
            return;
        }

        btn.innerText = "Membuat Music Card...";

        // Susun URL API Endpoint Nexray untuk Music Card
        const apiUrl = `https://api.nexray.eu.cc/canvas/musiccard?judul=${encodeURIComponent(title)}&nama=${encodeURIComponent(artist)}&image_url=${encodeURIComponent(finalImageUrl)}`;

        // Eksekusi pemanggilan gambar hasil
        resultImg.onload = function() {
            loader.style.display = "none";
            resultImg.style.display = "block";
            btnDownload.style.display = "inline-block";
            btnDownload.href = apiUrl; // Link siap didownload user
            
            btn.disabled = false;
            btn.innerText = "Generate Ulang";
        };

        resultImg.onerror = function() {
            loader.style.display = "none";
            placeholder.style.display = "block";
            placeholder.innerHTML = "<span style='color:#ff5a8a'>Gagal memproses kartu musik dari server. Silakan coba lagi.</span>";
            
            btn.disabled = false;
            btn.innerText = "Coba Lagi";
        };

        // Render hasil
        resultImg.src = apiUrl + "&_t=" + Date.now(); // Cache bypass
    }
</script>
