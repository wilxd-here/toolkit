<script>
    // ANIMASI BACKGROUND (Sama seperti sebelumnya)
    const canvas = document.getElementById('particles');
    const ctx = canvas.getContext('2d');
    function resize() { canvas.width = window.innerWidth; canvas.height = window.innerHeight; }
    resize(); window.addEventListener('resize', resize);
    const particles = [];
    for (let i = 0; i < 40; i++) {
        particles.push({
            x: Math.random() * canvas.width, y: Math.random() * canvas.height,
            size: Math.random() * 2 + 0.5, speedX: (Math.random() - 0.5) * 0.5, speedY: (Math.random() - 0.5) * 0.5,
            opacity: Math.random() * 0.5 + 0.1, color: Math.random() > 0.5 ? '#8a5aff' : '#ff8a5a'
        });
    }
    function animate() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        particles.forEach(p => {
            p.x += p.speedX; p.y += p.speedY;
            if (p.x < 0) p.x = canvas.width; if (p.x > canvas.width) p.x = 0;
            if (p.y < 0) p.y = canvas.height; if (p.y > canvas.height) p.y = 0;
            ctx.beginPath(); ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
            ctx.fillStyle = p.color; ctx.globalAlpha = p.opacity; ctx.fill();
        });
        ctx.globalAlpha = 1;
        requestAnimationFrame(animate);
    }
    animate();

    // LOGIKA GENERATOR FAKE IG
    async function generateFakeIG() {
        // Ambil Nilai Input
        const fileInput = document.getElementById('igImage');
        const username = document.getElementById('igUsername').value.trim();
        const posts = document.getElementById('igPosts').value.trim();
        const followers = document.getElementById('igFollowers').value.trim();
        const following = document.getElementById('igFollowing').value.trim();
        const bio = document.getElementById('igBio').value.trim();

        // Validasi
        if (!username || !posts || !followers || !following || !bio) {
            alert("Harap isi semua kolom teks!");
            return;
        }
        if (fileInput.files.length === 0) {
            alert("Harap upload Foto Profil (PP) terlebih dahulu!");
            return;
        }

        // Referensi UI
        const btn = document.getElementById('btnGenerate');
        const loader = document.getElementById('loader');
        const placeholder = document.getElementById('placeholder');
        const resultImg = document.getElementById('resultImg');
        const btnDownload = document.getElementById('btnDownload');

        // Loading state
        btn.disabled = true;
        placeholder.style.display = "none";
        resultImg.style.display = "none";
        btnDownload.style.display = "none";
        loader.style.display = "block";

        let finalImageUrl = "";

        // TAHAP 1: UPLOAD FOTO KE NEXA UPLOADER
        btn.innerText = "Mengunggah Foto...";
        const file = fileInput.files[0];
        const formData = new FormData();
        formData.append("files[]", file);

        try {
            const uploadResponse = await fetch("https://api.nexadev.my.id/uploder/", {
                method: "POST",
                body: formData
            });
            
            const uploadData = await uploadResponse.json();
            
            if (uploadData.success && uploadData.files && uploadData.files[0].url) {
                finalImageUrl = uploadData.files[0].url; 
            } else {
                throw new Error("Gagal mengunggah foto profil ke server.");
            }
        } catch (error) {
            console.error(error);
            alert("Gagal mengunggah gambar. Pastikan file valid atau coba beberapa saat lagi.");
            
            // Reset UI
            loader.style.display = "none";
            placeholder.style.display = "block";
            btn.disabled = false;
            btn.innerText = "Buat Profil";
            return;
        }

        // TAHAP 2: PROSES GENERATE GAMBAR FAKE IG
        btn.innerText = "Membuat Profil IG...";

        // Susun URL API Endpoint Fake IG
        const apiUrl = `https://apii.nexadev.my.id/fakeigprofile?ppurl=${encodeURIComponent(finalImageUrl)}&username=${encodeURIComponent(username)}&postingan=${encodeURIComponent(posts)}&mengikuti=${encodeURIComponent(following)}&pengikut=${encodeURIComponent(followers)}&bio=${encodeURIComponent(bio)}`;

        // Eksekusi pemanggilan gambar hasil
        resultImg.onload = function() {
            loader.style.display = "none";
            resultImg.style.display = "block";
            btnDownload.style.display = "inline-block";
            btnDownload.href = apiUrl; // Link untuk di-download pengguna
            
            btn.disabled = false;
            btn.innerText = "Buat Ulang Profil";
        };

        resultImg.onerror = function() {
            loader.style.display = "none";
            placeholder.style.display = "block";
            placeholder.innerHTML = "<span style='color:#ff5a8a'>Gagal memproses profil dari server API. Silakan coba lagi.</span>";
            
            btn.disabled = false;
            btn.innerText = "Coba Lagi";
        };

        // Render gambar (bypass cache bawaan browser agar selalu update)
        resultImg.src = apiUrl + "&_t=" + Date.now();
    }
</script>
