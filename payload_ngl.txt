    // ============================================================
    // XAERISOFT — NGL SPAM (THEME MERGED)
    // ============================================================

    const API_ENDPOINT = 'https://api.nexadev.my.id/tools/nglspam/';

    const usernameInput = document.getElementById('usernameInput');
    const messageInput = document.getElementById('messageInput');
    const countInput = document.getElementById('countInput');
    const startBtn = document.getElementById('startBtn');
    const logBox = document.getElementById('logBox');
    const stats = document.getElementById('stats');

    function addLog(msg, type = 'info') {
        const line = document.createElement('div');
        line.className = 'line ' + type;
        const time = new Date().toLocaleTimeString('id-ID', { hour12: false });
        line.textContent = `[${time}] ${msg}`;
        logBox.appendChild(line);
        logBox.scrollTop = logBox.scrollHeight;
    }

    function cleanUsername(input) {
        let username = input.trim();
        if (username.includes('ngl.link/')) {
            username = username.split('ngl.link/')[1];
            username = username.split('/')[0].split('?')[0];
        } else if (username.startsWith('@')) {
            username = username.substring(1);
        }
        return username;
    }

    startBtn.addEventListener('click', async () => {
        const username = cleanUsername(usernameInput.value);
        const message = messageInput.value.trim();
        const count = parseInt(countInput.value) || 5;

        if (!username) { alert('Masukkan username NGL target!'); return; }
        if (!message) { alert('Masukkan pesan!'); return; }
        if (count < 1 || count > 50) { alert('Jumlah min 1 max 50!'); return; }

        startBtn.disabled = true;
        startBtn.textContent = '⏳ Memproses...';
        logBox.classList.add('show');
        logBox.innerHTML = '';
        stats.innerHTML = '';

        addLog(`Target: @${username}`, 'info');
        addLog(`Pesan: ${message}`, 'info');
        addLog(`Jumlah: ${count}`, 'info');
        addLog('─────────────────────', 'info');

        try {
            const nglUrl = 'https://ngl.link/' + username;
            const url = `${API_ENDPOINT}?url=${encodeURIComponent(nglUrl)}&pesan=${encodeURIComponent(message)}&jumlah=${count}`;

            const response = await fetch(url);
            const data = await response.json();

            if (data.status === true) {
                addLog(`✓ ${count} pesan berhasil dikirim`, 'success');
                stats.innerHTML = `
                    <div>Sukses: <span>${count}</span></div>
                    <div>Gagal: <span>0</span></div>
                `;
            } else {
                addLog(`✗ Gagal: ${data.message || 'Unknown error'}`, 'error');
                stats.innerHTML = `
                    <div>Sukses: <span>0</span></div>
                    <div>Gagal: <span>${count}</span></div>
                `;
            }
        } catch (err) {
            addLog(`✗ Error: ${err.message}`, 'error');
        }

        startBtn.disabled = false;
        startBtn.textContent = '🚀 Kirim Sekarang';
    });

    console.log('XAERISOFT · NGL SPAM READY');

    // ============================================================
    // PARTICLE BACKGROUND ANIMATION (FROM PREVIOUS THEME)
    // ============================================================
    const canvas = document.getElementById('particles');
    const ctx = canvas.getContext('2d');

    function resize() {
        canvas.width = window.innerWidth;
        canvas.height = window.innerHeight;
    }
    resize();
    window.addEventListener('resize', resize);

    const particles = [];
    for (let i = 0; i < 40; i++) {
        particles.push({
            x: Math.random() * canvas.width,
            y: Math.random() * canvas.height,
            size: Math.random() * 2 + 0.5,
            speedX: (Math.random() - 0.5) * 0.5,
            speedY: (Math.random() - 0.5) * 0.5,
            opacity: Math.random() * 0.5 + 0.1,
            color: Math.random() > 0.5 ? '#8a5aff' : '#ff8a5a'
        });
    }

    function animate() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        particles.forEach(p => {
            p.x += p.speedX;
            p.y += p.speedY;
            if (p.x < 0) p.x = canvas.width;
            if (p.x > canvas.width) p.x = 0;
            if (p.y < 0) p.y = canvas.height;
            if (p.y > canvas.height) p.y = 0;
            ctx.beginPath();
            ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
            ctx.fillStyle = p.color;
            ctx.globalAlpha = p.opacity;
            ctx.fill();
        });
        ctx.globalAlpha = 1;
        requestAnimationFrame(animate);
    }
    animate();
