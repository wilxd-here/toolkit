<script>
    async function processDownload() {
        const inputUrl = document.getElementById('igUrl').value.trim();
        const btn = document.getElementById('btnFetch');
        const resultArea = document.getElementById('resultArea');
        
        if (!inputUrl) {
            alert('Tolong masukkan link Instagram terlebih dahulu!');
            return;
        }

        // Tampilan loading
        btn.innerText = '⏳ Sedang Memproses...';
        btn.disabled = true;
        resultArea.style.display = 'block';
        resultArea.innerHTML = '<p style="color:#8a7aaa; font-size:13px;">Sedang menghubungi server...</p>';

        try {
            // Asumsi API membutuhkan parameter ?url= untuk link Instagramnya
            // Jika format endpointnya berbeda (misal POST atau parameter lain), ubah di baris ini:
            const endpoint = `https://code.nexadev.my.id/?snippet=ig.js&url=${encodeURIComponent(inputUrl)}`;
            
            const response = await fetch(endpoint);
            const data = await response.json();

            // CATATAN PENTING: 
            // Karena saya tidak melihat langsung struktur JSON dari API nexadev, 
            // saya menggunakan asumsi umum 'data.url' atau 'data.media'.
            // Jika saat dijalankan error, kamu perlu mengecek isi console.log(data) 
            // dan menyesuaikan variabel 'videoUrl' di bawah ini.
            
            console.log("Respon API:", data); // Cek isi JSON di inspect element -> console

            // Misal API mengembalikan link download di dalam data.url atau data.data.url
            let downloadLink = data.url || (data.data && data.data.url); 

            if (downloadLink) {
                resultArea.innerHTML = `
                    <div class="result-card">
                        <p style="font-size:12px; color:#4aff9e; margin-bottom:10px;">✅ Berhasil Ditemukan!</p>
                        <a href="${downloadLink}" class="btn-download" target="_blank" rel="noopener noreferrer">⬇️ Download File</a>
                    </div>
                `;
            } else {
                resultArea.innerHTML = `<p class="error-msg">❌ Gagal mendapatkan link. Pastikan akun IG tidak diprivate atau coba link lain.</p>`;
            }

        } catch (error) {
            console.error(error);
            resultArea.innerHTML = `<p class="error-msg">⚠️ Terjadi kesalahan koneksi ke server API.</p>`;
        } finally {
            // Kembalikan tombol ke kondisi semula
            btn.innerText = 'Mulai Download';
            btn.disabled = false;
        }
    }
</script>
